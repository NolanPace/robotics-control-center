cmake_minimum_required(VERSION 3.16)
project(RobotSimulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(robot STATIC robot.cpp)
target_include_directories(robot PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(robot_simulator main.cpp)
target_link_libraries(robot_simulator PRIVATE robot)
enable_testing()
add_executable(robot_tests tests.cpp)
target_link_libraries(robot_tests PRIVATE robot)
add_test(NAME robot_tests COMMAND robot_tests)
